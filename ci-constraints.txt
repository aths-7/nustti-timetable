# CI 构建用的 pip 版本约束（由 workflow 通过 PIP_CONSTRAINT 生效）
#
# 背景：buildozer 传给 python-for-android 的 --requirements 里的版本号会被 p4a 丢弃，
# 于是 pip 会解析到 charset-normalizer 3.5+；该版本在 PyPI 上发布了 android 预编译
# wheel（如 charset_normalizer-3.5.1-cp314-cp314-android_24_arm64_v8a.whl），p4a 的
# 平台解析阶段会选中它，但真正执行 `pip install --target <dist>` 时没有带 --platform，
# 平台标签不匹配 → 直接报 "is not a supported wheel on this platform" 构建失败。
#
# 3.3.2 只有 py3-none-any 纯 Python wheel，不含 android wheel，安装必然成功。
charset-normalizer==3.3.2

# 与 buildozer.spec 中声明保持一致，避免 pip 解析到更新的 requests 引入新依赖
requests==2.32.3
